CREATE DATABASE local_library;
USE local_library ;
CREATE TABLE books(
  book_id INT AUTO_INCREMENT NOT NULL,
  title VARCHAR(50),
    PRIMARY KEY(book_id));

USE local_library ;
CREATE TABLE borrowers(
  book_id INT PRIMARY KEY NOT NULL,
  title VARCHAR(50),
    FOREIGN KEY(book_id) REFERENCES books(book_id));
