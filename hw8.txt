CREATE DATABASE company;
USE company;
CREATE TABLE employees(
    id INT AUTO_INCREMENT NOT NULL,
    name VARCHAR(50),
    PRIMARY KEY(id));
CREATE TABLE departments(
    emp_id INT NOT NULL,
    department_name VARCHAR(150),
    PRIMARY KEY(emp_id) );
